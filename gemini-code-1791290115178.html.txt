            "④ [해설] 4번 선택지 상세 분석 및 정답/오답 사유입니다.",
            "⑤ [해설] 5번 선택지 상세 분석 및 정답/오답 사유입니다."
        ]
    });
}

let currentIndex = 0;
let userAnswers = new Array(questions.length).fill(null);

function initJumpSelect() {
    const select = document.getElementById("jump-select");
    select.innerHTML = "";
    questions.forEach((q, idx) => {
        const opt = document.createElement("option");
        opt.value = idx;
        opt.innerText = `${idx + 1}번 문제`;
        select.appendChild(opt);
    });
}

function loadQuestion() {
    const q = questions[currentIndex];
    document.getElementById("question-number").innerText = `문제 ${currentIndex + 1} / ${questions.length}`;
    document.getElementById("question-text").innerText = q.question;
    document.getElementById("progress-bar").style.width = `${((currentIndex + 1) / questions.length) * 100}%`;
    document.getElementById("jump-select").value = currentIndex;

    const optionsContainer = document.getElementById("options-list");
    optionsContainer.innerHTML = "";

    const expBox = document.getElementById("explanation-box");
    expBox.classList.remove("show");
    expBox.innerHTML = "";

    q.options.forEach((opt, idx) => {
        const btn = document.createElement("button");
        btn.className = "option-btn";
        btn.innerText = `${idx + 1}. ${opt}`;
        btn.onclick = () => selectOption(idx);

        if (userAnswers[currentIndex] !== null) {
            btn.disabled = true;
            if (idx === q.answer) {
                btn.classList.add("selected-correct");
            }
            if (userAnswers[currentIndex] === idx && idx !== q.answer) {
                btn.classList.add("selected-incorrect");
            }
        }

        optionsContainer.appendChild(btn);
    });

    if (userAnswers[currentIndex] !== null) {
        let html = `<div class="explanation-title">💡 전체 해설</div><div>${q.explanation}</div><hr style="border:0; border-top:1px solid #e2e8f0; margin: 8px 0;">`;
        q.optionExplanations.forEach((exp) => {
            html += `<div class="option-exp-item">${exp}</div>`;
        });
        expBox.innerHTML = html;
        expBox.classList.add("show");
    }

    document.getElementById("prev-btn").disabled = currentIndex === 0;
    document.getElementById("next-btn").innerText = currentIndex === questions.length - 1 ? "결과 보기" : "다음";
}

function selectOption(index) {
    if (userAnswers[currentIndex] !== null) return;
    userAnswers[currentIndex] = index;
    loadQuestion();
}

function nextQuestion() {
    if (currentIndex < questions.length - 1) {
        currentIndex++;
        loadQuestion();
    } else {
        showResults();
    }
}

function prevQuestion() {
    if (currentIndex > 0) {
        currentIndex--;
        loadQuestion();
    }
}

function jumpToQuestion(index) {
    currentIndex = parseInt(index);
    loadQuestion();
}

function showResults() {
    let score = 0;
    userAnswers.forEach((ans, idx) => {
        if (ans === questions[idx].answer) score++;
    });

    const body = document.getElementById("quiz-body");
    body.innerHTML = `
        <div class="results-container">
            <h2>모의고사 풀이 완료!</h2>
            <div class="score-circle">${score}점</div>
            <p>총 ${questions.length}문항 중 <strong>${score}문항</strong>을 맞히셨습니다.</p>
            <button class="nav-btn" onclick="location.reload()">다시 풀기</button>
        </div>
    `;
    document.querySelector(".footer-nav").style.display = "none";
}

initJumpSelect();
loadQuestion();
